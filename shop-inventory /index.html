let inventory = []; let transactions = []; document.addEventListener("DOMContentLoaded", () => { fetchStockData(); document.getElementById("sales-form").addEventListener("submit", function(e) { e.preventDefault(); recordLiveSale(); }); }); // Fetch stocks from Python backend / Google Sheet function fetchStockData() { fetch('/api/get-stocks') .then(res => res.json()) .then(data => { inventory = data.map((item, index) => ({ id: item["PRODUCT ID"] || index + 1, name: item["PRODUCT NAME"], stock: parseInt(item["STOCK"] || 0), sold: 0, price: 100 // default or fetch if you add price column })); populateProductDropdown(); renderDashboard(); }); } function populateProductDropdown() { const select = document.getElementById("product-select"); select.innerHTML = ''; inventory.forEach(item => { const option = document.createElement("option"); option.value = item.id; option.textContent = item.name; select.appendChild(option); }); } function recordLiveSale() { const productId = parseInt(document.getElementById("product-select").value); const quantity = parseInt(document.getElementById("quantity-input").value); const product = inventory.find(item => item.id === productId); if (!product || !quantity) return; // Send data to Python backend to update Google Sheet instantly fetch('/api/record-sale', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ productName: product.name, qtySold: quantity, notAvailable: "" }) }) .then(res => res.json()) .then(response => { product.stock = response.stockAfterSale; product.sold += quantity; transactions.push({ productId: product.id, quantity: quantity, timestamp: new Date() }); renderDashboard(); document.getElementById("sales-form").reset(); }); }
